// The flashlight that lights dark mode, on both pages. It follows a mouse or pen, and the page is marked data-lit so the // line in the margin and the 404's way out can hide in the dark. On touch screens and with reduced motion it stays put // and nothing hides. Its battery: after two minutes of dark mode it narrows, cools and starts to flicker over the next // three, until a quick shake of the mouse brings it back. (It used to be a button whose choice was saved, so that is cleared.) (function () { var root = document.documentElement, light = document.querySelector('.flashlight'); if (!light) return; try { localStorage.removeItem('flashlight'); } catch (e) {} light.hidden = false; var fine = window.matchMedia('(hover: hover) and (pointer: fine)'), still = window.matchMedia('(prefers-reduced-motion: reduce)'), system = window.matchMedia('(prefers-color-scheme: dark)'); if (!fine.matches || still.matches) return; root.dataset.lit = ''; // The line in the margin stays hidden in the page itself until now, so a stale or missing stylesheet can never show it. var ink = document.querySelector('.ink'); if (ink) { ink.hidden = false; if (getComputedStyle(ink).position !== 'fixed') ink.hidden = true; // Its styles never arrived, so keep it out of sight. } var out = document.querySelector('.way-out'), frame = 0, x, y, lx = null, ly = null, lt = 0, shake = 0, charge = 1, lit = 0; function dark() { return root.dataset.theme ? root.dataset.theme === 'dark' : system.matches; } function power() { root.style.setProperty('--fr', Math.round(170 + 170 * charge) + 'px'); root.style.setProperty('--fw', (0.01 + 0.04 * charge).toFixed(3)); } document.addEventListener('pointermove', function (e) { if (e.pointerType === 'touch') return; x = e.clientX; y = e.clientY; var t = performance.now(); if (lx !== null) { var v = Math.sqrt((x - lx) * (x - lx) + (y - ly) * (y - ly)) / Math.max(1, t - lt); shake = shake * 0.92 + (v > 2.2 ? v : 0); if (shake > 40 && charge < 0.98) { charge = 1; lit = 0; shake = 0; power(); } } lx = x; ly = y; lt = t; if (!frame) frame = requestAnimationFrame(function () { frame = 0; root.style.setProperty('--fx', x + 'px'); root.style.setProperty('--fy', y + 'px'); if (out) { var r = out.getBoundingClientRect(); out.style.setProperty('--wx', (x - r.left) + 'px'); out.style.setProperty('--wy', (y - r.top) + 'px'); } }); }, { passive: true }); setInterval(function () { if (!dark() || document.hidden) return; lit++; charge = lit < 120 ? 1 : Math.max(0, 1 - (lit - 120) / 180); power(); }, 1000); (function flicker() { var low = charge < 0.45 && dark(); if (low) { root.style.setProperty('--fd', '0.22'); setTimeout(function () { root.style.setProperty('--fd', '0'); }, 70 + Math.random() * 90); } setTimeout(flicker, low ? 600 + Math.random() * 2200 * (charge + 0.2) : 1000); })(); })();
